function doPost(e) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var p = e.parameter; var truckNo = (p.truckNo || "").trim(); var driverName = (p.driverName || "").trim(); var soNo = (p.soNo || "").trim(); var ocName = (p.ocName || "").trim(); var gateIn = p.gateIn || ""; var bayIn = p.bayIn || ""; var bayOut = p.bayOut || ""; var gateOut = p.gateOut || ""; var delayReason = p.delayReason || ""; var ocNotes = p.ocNotes || ""; // Calculate times in minutes var parkingTime = calculateDiffInMinutes(gateIn, bayIn); var loadingTime = calculateDiffInMinutes(bayIn, bayOut); var totalTRT = calculateDiffInMinutes(gateIn, gateOut); var data = sheet.getDataRange().getValues(); var foundIndex = -1; // Search for existing active record (without Gate Out) matching Truck No or SO No for (var i = 1; i < data.length; i++) { var rowTruck = String(data[i][0]).trim(); var rowSO = String(data[i][2]).trim(); var rowGateOut = data[i][7]; if ((rowTruck === truckNo || (soNo !== "" && rowSO === soNo)) && (!rowGateOut || rowGateOut === "")) { foundIndex = i + 1; // 1-based index break; } } if (foundIndex > 0) { // Update existing record if (driverName) sheet.getRange(foundIndex, 2).setValue(driverName); if (soNo) sheet.getRange(foundIndex, 3).setValue(soNo); if (ocName) sheet.getRange(foundIndex, 4).setValue(ocName); if (gateIn) sheet.getRange(foundIndex, 5).setValue(gateIn); if (bayIn) sheet.getRange(foundIndex, 6).setValue(bayIn); if (bayOut) sheet.getRange(foundIndex, 7).setValue(bayOut); if (gateOut) sheet.getRange(foundIndex, 8).setValue(gateOut); sheet.getRange(foundIndex, 9).setValue(parkingTime); sheet.getRange(foundIndex, 10).setValue(loadingTime); sheet.getRange(foundIndex, 11).setValue(totalTRT); sheet.getRange(foundIndex, 12).setValue(delayReason); sheet.getRange(foundIndex, 13).setValue(ocNotes); } else { // Append new record sheet.appendRow([ truckNo, driverName, soNo, ocName, gateIn, bayIn, bayOut, gateOut, parkingTime, loadingTime, totalTRT, delayReason, ocNotes ]); } return ContentService.createTextOutput(JSON.stringify({ status: "success" })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function calculateDiffInMinutes(startStr, endStr) { if (!startStr || !endStr) return ""; var start = new Date(startStr); var end = new Date(endStr); if (isNaN(start.getTime()) || isNaN(end.getTime())) return ""; var diff = Math.round((end - start) / (1000 * 60)); return diff >= 0 ? diff : 0; } function doGet(e) { e = e || { parameter: {} }; var action = e.parameter.action || ""; if (action === "getActive") { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = sheet.getDataRange().getValues(); var activeTrucks = []; for (var i = 1; i < data.length; i++) { var rowGateOut = data[i][7]; if (!rowGateOut || rowGateOut === "") { activeTrucks.push({ truckNo: data[i][0], driverName: data[i][1], soNo: data[i][2], ocName: data[i][3], gateIn: data[i][4], bayIn: data[i][5], bayOut: data[i][6], gateOut: data[i][7], parkingTime: data[i][8], loadingTime: data[i][9], totalTRT: data[i][10], delayReason: data[i][11], ocNotes: data[i][12] }); } } return ContentService.createTextOutput(JSON.stringify(activeTrucks)) .setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput("TRT API Active"); }
